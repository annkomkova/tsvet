Верстаем «Цветочный» по кускам

Заглушка главной страницы учебного проекта, разобранная на шаги в том порядке, в котором мы верстаем её на паре. Каждый кусок сразу с адаптивом. После любого шага страница открывается без ошибок, так что остановиться можно где угодно.

  • HTML — по слотамВ разметке стоят комментарии <!-- ↓ 2.3 … -->. Кусок с этим номером вставляем вместо комментария.
  • CSS — в конец файлаСтили каждого куска дописываем в конец style.css по порядку шагов.
  • JS — как в курсеВызов функции дописываем в список в начале main.js, саму функцию — в конец файла. Только let, querySelector, classList, события, setInterval.
  • SVG — отдельноКаждая картинка — свой кусок, даже если такой же цветок уже был. Копируйте целиком.
  • Никаких data-атрибутовJS находит всё по классам: split--chars, magnetic, theme--dark и т. д.
  • FormspreeФорма отправляется без JS. Перед показом замените YOUR_FORM_ID в шаге 6.2 на ID своей формы.

function headerScroll() {
  let header = document.querySelector('.header')
  let lastScroll = 0

  window.addEventListener('scroll', () => {
    let scroll = window.scrollY

    if (scroll > 20) {
      header.classList.add('header--scrolled')
    } else {
      header.classList.remove('header--scrolled')
    }

    if (scroll > 160 && scroll > lastScroll && !header.classList.contains('header--menu')) {
      header.classList.add('header--hidden')
    } else {
      header.classList.remove('header--hidden')
    }

    lastScroll = scroll
  })
}

// 2.2 Разбиваем текст на буквы и слова
function splitText() {
  let elements = document.querySelectorAll('.split')

  elements.forEach((element) => {
    let words = element.innerText.split(' ')
    let html = ''
    let index = 0
    let delay = 0

    if (element.classList.contains('split--delay')) {
      delay = 300
    }

    words.forEach((word) => {
      html += '<span class="split__word">'

      if (element.classList.contains('split--chars')) {
        for (let i = 0; i < word.length; i++) {
          html += `<span class="split__item" style="transition-delay: ${delay + index * 45}ms">${word[i]}</span>`
          index++
        }
      } else {
        html += `<span class="split__item" style="transition-delay: ${delay + index * 45}ms">${word}</span>`
        index++
      }

      html += '</span> '
    })

    element.innerHTML = html
  })
}

function checkElements() {
  if (!document.body.classList.contains('page--loading')) {
    let elements = document.querySelectorAll('.split, .reveal, .fade')

    elements.forEach((element) => {
      let rect = element.getBoundingClientRect()

      if (rect.top < window.innerHeight * 0.9) {
        if (element.classList.contains('split')) {
          element.classList.add('split--visible')
        }
        if (element.classList.contains('reveal')) {
          element.classList.add('reveal--visible')
        }
        if (element.classList.contains('fade')) {
          element.classList.add('fade--visible')
        }
      }
    })
  }
}

3.2

Движение от скролла

Что проговорить

  • Положение ленты храним в процентах от ширины копии: от 0 до −100. Каждые 16 мс (≈60 раз в секунду) сдвигаем его через setInterval — как прогресс-бар в курсе. Вышли за −100 — прибавляем 100, и лента незаметно начинается заново.
  • Нижняя лента едет зеркально: её положение −100 минус положение верхней.
  • На скролле сравниваем новую прокрутку с прошлой: листаем вниз — ленты едут в одну сторону, вверх — в другую. Чем быстрее листаем, тем больше скорость и перекос секции.
  • Когда скролл остановился, возвращаем всё как было. Для этого каждый раз отменяем прошлый таймер (clearTimeout) и ставим новый на 150 мс — он сработает, только когда скролл затихнет.

// 3. Бегущая строка
function moveMarquee() {
  let marquee = document.querySelector('.marquee')
  let rows = document.querySelectorAll('.marquee__row')
  let position = 0
  let direction = 1
  let speed = 1
  let lastScroll = window.scrollY
  let timer

  window.addEventListener('scroll', () => {
    let delta = window.scrollY - lastScroll
    lastScroll = window.scrollY

    if (delta > 0) {
      direction = 1
    } else if (delta < 0) {
      direction = -1
    }

    speed = 1 + Math.min(Math.abs(delta) / 10, 4)

    let skew = Math.max(-8, Math.min(8, delta * 0.3))
    marquee.style.transform = `skewX(${skew}deg)`

    clearTimeout(timer)
    timer = setTimeout(() => {
      marquee.style.transform = 'skewX(0deg)'
      speed = 1
    }, 150)
  })

  setInterval(() => {
    position = position - 0.05 * speed * direction

    if (position < -100) {
      position = position + 100
    }
    if (position > 0) {
      position = position - 100
    }

    rows.forEach((row) => {
      let tracks = row.querySelectorAll('.marquee__track')
      let x = position

      if (row.classList.contains('marquee__row--reverse')) {
        x = -100 - position
      }

      tracks.forEach((track) => {
        track.style.transform = `translateX(${x}%)`
      })
    })
  }, 16)
}

// 5.2 3D-наклон карточки за мышью
function tiltCard() {
  let link = document.querySelector('.card__link')
  let media = document.querySelector('.card__media')

  link.addEventListener('mousemove', (event) => {
    let rect = media.getBoundingClientRect()
    let x = (event.clientX - rect.left) / rect.width - 0.5
    let y = (event.clientY - rect.top) / rect.height - 0.5

    media.style.transform = `perspective(1000px) rotateX(${-y * 10}deg) rotateY(${x * 10}deg)`
  })

  link.addEventListener('mouseleave', () => {
    media.style.transform = ''
  })
}

5.4

Пятно-превью за курсором

Что проговорить

  • Пятно зафиксировано (position: fixed) и на mousemove едет в точку курсора. transition на transform заставляет его немного отставать — получается мягкое движение.
  • Пока курсор не над строкой списка, пятно сжато в scale(0). Над строкой — добавляем модификатор preview--active.
  • Цвет: у пятна есть модификаторы preview--pink, preview--lilac, preview--lime, preview--peach. Названия лежат в массиве colors в том же порядке, что строки списка, поэтому по номеру строки (index в forEach) берём нужный цвет.
  • preview.className перезаписывает сразу все классы элемента: так старый цвет не остаётся висеть.
  • mouseenter и mouseleave похожи на mouseover и mouseout, но не срабатывают повторно, когда мышь переходит между вложенными элементами строки.

// 5.4 Пятно-превью за курсором
function showPreview() {
  let preview = document.querySelector('.preview')
  let items = document.querySelectorAll('.soon__item')
  let colors = ['pink', 'lilac', 'lime', 'peach']

  document.addEventListener('mousemove', (event) => {
    preview.style.transform = `translate(${event.clientX}px, ${event.clientY}px)`
  })

  items.forEach((item, index) => {
    item.addEventListener('mouseenter', () => {
      preview.className = `preview preview--active preview--${colors[index]}`
    })

    item.addEventListener('mouseleave', () => {
      preview.classList.remove('preview--active')
    })
  })
}

6.2

Поля с плавающими подписями

Что проговорить

  • Форма отправляется на Formspree самым простым способом: атрибуты action и method. Создайте форму на formspree.io и замените YOUR_FORM_ID в action. После отправки Formspree покажет свою страницу «Спасибо» со ссылкой назад.
  • Проверку делает браузер: required — поле обязательно, type="email" — нужен email. Пока поле заполнено неправильно, форма не отправится, а браузер покажет подсказку.
  • :user-invalid срабатывает, только когда человек уже потрогал поле и ввёл что-то не то — тогда линия под полем краснеет. Пустая форма при загрузке красной не будет.
  • Плавающая подпись: у инпута placeholder=" " (пробел). Пока поле пустое, placeholder «показан» — срабатывает :placeholder-shown. Когда поле в фокусе или заполнено, подпись уезжает вверх и уменьшается. Поэтому label стоит ПОСЛЕ input — чтобы работал селектор соседа +.
  • Линия под полем: ::after у form__field растягивается scaleX(0 → 1) при :focus-within.
  • Скрытое поле _gotcha — ловушка для спам-ботов, _subject — тема письма в Formspree.

Свой курсор вместо системного: белый кружок в режиме смешивания difference (инвертирует всё, что под ним), увеличивается над ссылками и превращается в розовый круг с подписью над карточкой, бейджем и строками «скоро». Плюс магнитные кнопки.

9.1

Курсор с подписями

Что проговорить

  • Курсор показываем только там, где есть мышь: медиазапрос (hover: hover) and (pointer: fine). На телефонах и планшетах он не появится, и системный курсор останется.
  • Как setCursor() в курсе: на mousemove двигаем курсор в точку мыши. Небольшое отставание даёт transition на transform.
  • Над ссылками и кнопками (mouseover) добавляем cursor--hover, при уходе (mouseout) убираем.
  • Подписи вешаем внутренней функцией addCursorLabel(selector, text): она находит элементы по селектору и на наведение пишет в курсор нужный текст. Одна функция — три вызова: карточка «Читать», бейдж «Скоро!», строки списка «Скоро».

function moveCursor() {
  let cursor = document.querySelector('.cursor')
  let label = document.querySelector('.cursor__label')
  let links = document.querySelectorAll('a, button, label')

  document.addEventListener('mousemove', (event) => {
    cursor.style.transform = `translate(${event.clientX}px, ${event.clientY}px)`
  })

  links.forEach((link) => {
    link.addEventListener('mouseover', () => {
      cursor.classList.add('cursor--hover')
    })
    link.addEventListener('mouseout', () => {
      cursor.classList.remove('cursor--hover')
    })
  })

  addCursorLabel('.card__link', 'Читать')
  addCursorLabel('.badge', 'Скоро!')
  addCursorLabel('.soon__item', 'Скоро')

  function addCursorLabel(selector, text) {
    let elements = document.querySelectorAll(selector)

        label.innerText = text
        cursor.classList.add('cursor--label')
      })
      element.addEventListener('mouseout', () => {
        cursor.classList.remove('cursor--label')
      })
    })
  }
}

function magneticButtons() {
  let buttons = document.querySelectorAll('.magnetic')

  buttons.forEach((button) => {
    button.addEventListener('mousemove', (event) => {
      let rect = button.getBoundingClientRect()
      let x = event.clientX - (rect.left + rect.width / 2)
      let y = event.clientY - (rect.top + rect.height / 2)

    })

      button.style.transform = ''
    })
  })
}
